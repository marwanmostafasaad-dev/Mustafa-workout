    function loadSplitData() {
      try {
        const saved = localStorage.getItem(STORAGE_KEY);
        if (saved) {
          const parsed = JSON.parse(saved);
          if (parsed && typeof parsed === "object" && Object.keys(parsed).length > 0) {
            return parsed;
          }
        }
      } catch (e) {
        console.warn("Could not read localStorage:", e);
      }
      return JSON.parse(JSON.stringify(DEFAULT_SPLIT_DATA));
    }

    function saveSplitData() {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(splitData));
      } catch (e) {
        console.warn("Could not save to localStorage:", e);
      }
    }

    // =====================================================================
    // BULLETPROOF YOUTUBE LINK & ID PARSER
    // =====================================================================
    function extractYouTubeId(input) {
      if (!input) return "";
      const trimmed = String(input).trim();

      // 1. Direct 11-character Video ID check
      if (/^[a-zA-Z0-9_-]{11}$/.test(trimmed)) {
        return trimmed;
      }

      // 2. Standard URL parsing
      try {
        const urlStr = /^https?:\/\//i.test(trimmed) ? trimmed : `https://${trimmed}`;
        const urlObj = new URL(urlStr);
        const hostname = urlObj.hostname.replace(/^www\.|^m\./i, "").toLowerCase();

        if (hostname === "youtu.be") {
          const id = urlObj.pathname.split("/").filter(Boolean)[0];
          if (id && /^[a-zA-Z0-9_-]{11}$/.test(id)) return id;
        }

        if (hostname.endsWith("youtube.com") || hostname.endsWith("youtube-nocookie.com")) {
          const vParam = urlObj.searchParams.get("v");
          if (vParam && /^[a-zA-Z0-9_-]{11}$/.test(vParam)) return vParam;

          const segments = urlObj.pathname.split("/").filter(Boolean);
          if (segments.length >= 2 && ["shorts", "embed", "v", "live"].includes(segments[0])) {
            const pathId = segments[1];
            if (/^[a-zA-Z0-9_-]{11}$/.test(pathId)) return pathId;
          }
        }
      } catch (e) {
        // Fallback to regex below
      }

      // 3. Comprehensive Regex Fallback
      const regexList = [
        /(?:youtube\.com\/(?:watch\?.*v=|embed\/|v\/|shorts\/|live\/)|youtu\.be\/)([a-zA-Z0-9_-]{11})/,
        /[?&]v=([a-zA-Z0-9_-]{11})/
      ];
      for (const rx of regexList) {
        const match = trimmed.match(rx);
        if (match && match[1]) return match[1];
      }

      return "";
    }

    // Live Preview Updater inside Exercise Editor Modal
    function updateVideoPreviewBadge() {
      const rawVal = document.getElementById("inputExVideo").value;
      const previewBox = document.getElementById("videoPreviewBox");
      const validState = document.getElementById("videoValidState");
      const invalidState = document.getElementById("videoInvalidState");
      const thumbImg = document.getElementById("videoPreviewThumb");
      const idLabel = document.getElementById("parsedIdLabel");

      if (!rawVal.trim()) {
        previewBox.classList.add("hidden");
        return;
      }

      previewBox.classList.remove("hidden");
      const parsedId = extractYouTubeId(rawVal);

      if (parsedId) {
        validState.classList.remove("hidden");
        invalidState.classList.add("hidden");
        idLabel.textContent = parsedId;
        thumbImg.src = `https://img.youtube.com/vi/${parsedId}/mqdefault.jpg`;

        if (/\/shorts\//i.test(rawVal)) {
          document.getElementById("inputExRatio").value = "9:16";
        }
      } else {
        validState.classList.add("hidden");
        invalidState.classList.remove("hidden");
      }
    }

    // Toast Notification Helper
    function showToast(message, type = "success") {
      const container = document.getElementById("toastContainer");
      const toast = document.createElement("div");
      const accentBorder = type === "error" ? "border-rose-500/50" : "border-cobalt-500/50";
      toast.className = `pointer-events-auto bg-graphite border ${accentBorder} text-[#f8fafc] px-4 py-3 rounded-xl shadow-2xl text-xs sm:text-sm font-medium flex items-center gap-2.5 transform transition-all duration-300 translate-y-2 opacity-0`;

      toast.innerHTML = `
        <span class="w-2 h-2 rounded-full ${type === 'error' ? 'bg-rose-500' : 'bg-cobalt-400'} shrink-0"></span>
        <span>${escapeHTML(message)}</span>
      `;
      container.appendChild(toast);

      requestAnimationFrame(() => {
        toast.classList.remove("translate-y-2", "opacity-0");
      });

      setTimeout(() => {
        toast.classList.add("translate-y-2", "opacity-0");
        setTimeout(() => toast.remove(), 300);
      }, 3400);
    }

    // Render Day Tabs
    function renderTabs() {
      const container = document.getElementById("dayTabsContainer");
      container.innerHTML = "";
      const days = Object.keys(splitData);

      if (!days.includes(activeDay) && days.length > 0) {
        activeDay = days[0];
      }

      days.forEach(day => {
        const isActive = day === activeDay;
        const wrapper = document.createElement("div");
        wrapper.className = "flex items-center shrink-0";

        const btn = document.createElement("button");
        btn.className = isActive
          ? "px-4 py-2.5 rounded-xl bg-gradient-to-r from-cobalt-500 to-violetNeon-500 text-white font-extrabold text-xs sm:text-sm shadow-lg shadow-cobalt-500/25 transition cursor-pointer"
          : "px-4 py-2.5 rounded-xl bg-graphite hover:bg-graphiteHover border border-metallic text-slate-400 hover:text-[#f8fafc] font-semibold text-xs sm:text-sm transition cursor-pointer";
        btn.textContent = day;
        btn.onclick = () => {
          activeDay = day;
          renderTabs();
          renderExercises();
        };
        wrapper.appendChild(btn);

        // Delete Day Button in Edit Mode
        if (isEditMode && days.length > 1) {
          const delBtn = document.createElement("button");
          delBtn.className = "ml-1 p-1.5 rounded-lg bg-void hover:bg-rose-950 text-slate-500 hover:text-rose-400 border border-metallic text-xs cursor-pointer";
          delBtn.title = `Remove ${day}`;
          delBtn.innerHTML = "&times;";
          delBtn.onclick = (e) => {
            e.stopPropagation();
            delete splitData[day];
            saveSplitData();
            activeDay = Object.keys(splitData)[0];
            renderTabs();
            renderExercises();
            showToast(`Removed "${day}" from split`);
          };
          wrapper.appendChild(delBtn);
        }

        container.appendChild(wrapper);
      });
    }

    // Move Exercise Up/Down in Edit Mode
    function moveExercise(index, direction) {
      const list = splitData[activeDay];
      if (!list) return;
      const targetIndex = index + direction;
      if (targetIndex < 0 || targetIndex >= list.length) return;

      const temp = list[index];
      list[index] = list[targetIndex];
      list[targetIndex] = temp;

      saveSplitData();
      renderExercises();
    }

    function escapeHTML(str) {
      return String(str || "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;");
    }

    function openVideoModal(exercise) {
      const modal = document.getElementById("videoModal");
      const iframe = document.getElementById("youtubeIframe");
      const dayBadge = document.getElementById("modalDayBadge");
      const titleEl = document.getElementById("modalExerciseName");
      const setsBadge = document.getElementById("modalSetsBadge");
      const cuesEl = document.getElementById("modalCuesText");
      const fallbackLink = document.getElementById("modalFallbackLink");

      const cleanId = extractYouTubeId(exercise.videoId);

      if (embedCheckTimer) {
        clearTimeout(embedCheckTimer);
        embedCheckTimer = null;
      }
      hideEmbedFallback();

      dayBadge.textContent = activeDay;
      titleEl.textContent = exercise.name;
      setsBadge.textContent = exercise.sets;
      cuesEl.textContent = exercise.cues || "Control the eccentric, pause briefly at peak stretch/contraction, and maintain strict posture.";

      fallbackLink.href = cleanId
        ? `https://www.youtube.com/watch?v=${cleanId}`
        : "https://www.youtube.com";

      setAspectRatio(exercise.aspect || "16:9");

      if (cleanId) {
        iframe.onload = () => {
          if (embedCheckTimer) clearTimeout(embedCheckTimer);
        };
        iframe.onerror = () => {
          showEmbedFallback(cleanId);
        };
        // Clean standard YouTube embed URL without conflicting origin/referrer parameters
        iframe.src = `https://www.youtube.com/embed/${cleanId}?playsinline=1&rel=0`;

        // If a sandboxed or restricted frame prevents iframe load completion, surface direct watch CTA
        embedCheckTimer = setTimeout(() => {
          try {
            const rect = iframe.getBoundingClientRect();
            if (rect.width === 0 || rect.height === 0) {
              showEmbedFallback(cleanId);
            }
          } catch (_) {
            // Keep clean iframe view if accessible
          }
        }, 2500);
      } else {
        iframe.src = "";
        showEmbedFallback("");
      }

      modal.classList.remove("hidden");
      modal.classList.add("flex");
      document.body.style.overflow = "hidden";
    }

    function closeVideoModal() {
      const modal = document.getElementById("videoModal");
      const iframe = document.getElementById("youtubeIframe");

      if (embedCheckTimer) {
        clearTimeout(embedCheckTimer);
        embedCheckTimer = null;
      }
      hideEmbedFallback();

      modal.classList.add("hidden");
      modal.classList.remove("flex");
      iframe.onload = null;
      iframe.onerror = null;
      iframe.src = "";
      document.body.style.overflow = "";
    }

        addBtn.classList.remove("flex");
        label.textContent = "Customize Split";
        toggleBtn.classList.remove("border-cobalt-500", "text-white");
      }

      renderTabs();
      renderExercises();
    }

    function openExerciseFormModal(index = -1, focusVideo = false) {
      const modal = document.getElementById("exerciseFormModal");
      const title = document.getElementById("formModalTitle");
      const idxInput = document.getElementById("editExerciseIndex");
      const nameInput = document.getElementById("inputExName");
      const setsInput = document.getElementById("inputExSets");
      const ratioInput = document.getElementById("inputExRatio");
      const videoInput = document.getElementById("inputExVideo");
      const cuesInput = document.getElementById("inputExCues");

      idxInput.value = index;

      if (index >= 0 && splitData[activeDay][index]) {
        const ex = splitData[activeDay][index];
        title.textContent = focusVideo ? `Update Video Link — ${ex.name}` : "Edit Movement";
        nameInput.value = ex.name;
        setsInput.value = ex.sets;
        ratioInput.value = ex.aspect || "16:9";
        videoInput.value = ex.videoId;
        cuesInput.value = ex.cues || "";
      } else {
        title.textContent = `Add Movement to ${activeDay}`;
        nameInput.value = "";
        setsInput.value = activeDay.toLowerCase().includes("lower") ? "2 sets x 6-8 reps" : "3 sets x 8-10 reps";
        ratioInput.value = "16:9";
        videoInput.value = "";
        cuesInput.value = "";
      }

      updateVideoPreviewBadge();

      modal.classList.remove("hidden");
      modal.classList.add("flex");

      if (focusVideo) {
        videoInput.focus();
        videoInput.select();
      } else {
        nameInput.focus();
      }
    }

    function closeExerciseFormModal() {
      const modal = document.getElementById("exerciseFormModal");
      modal.classList.add("hidden");
      modal.classList.remove("flex");
    }

    function saveExerciseForm(e) {
      e.preventDefault();
      const index = parseInt(document.getElementById("editExerciseIndex").value, 10);
      const name = document.getElementById("inputExName").value.trim();
      const sets = document.getElementById("inputExSets").value.trim();
      const aspect = document.getElementById("inputExRatio").value;
      const rawVideo = document.getElementById("inputExVideo").value.trim();
      const cues = document.getElementById("inputExCues").value.trim();

      const videoId = extractYouTubeId(rawVideo);
      if (!videoId) {
        showToast("Invalid YouTube URL or ID. Paste a valid YouTube link or 11-character ID.", "error");
        return;
      }

      const entry = { name, sets, videoId, aspect, cues };

      if (!splitData[activeDay]) splitData[activeDay] = [];

      if (index >= 0) {
        splitData[activeDay][index] = entry;
        showToast(`Saved "${name}" (Video ID: ${videoId})`);
      } else {
        splitData[activeDay].push(entry);
        showToast(`Added "${name}" to ${activeDay}`);
      }

      saveSplitData();
      closeExerciseFormModal();
      renderExercises();
    }

    function deleteExercise(index) {
      const removed = splitData[activeDay].splice(index, 1);
      saveSplitData();
      renderExercises();
      if (removed[0]) {
        showToast(`Removed "${removed[0].name}"`);
      }
    }

    function saveNewDay(e) {
      e.preventDefault();
      const name = document.getElementById("inputNewDayName").value.trim();
      if (!name) return;

      if (splitData[name]) {
        showToast(`"${name}" already exists in your split.`, "error");
        return;
      }

      splitData[name] = [];
      activeDay = name;
      saveSplitData();
      closeAddDayModal();
      renderTabs();
      renderExercises();
      showToast(`Created day "${name}"`);
    }

    // Rename Day Modal Handlers
    function openRenameDayModal() {
      const modal = document.getElementById("renameDayModal");
      document.getElementById("renameTargetLabel").textContent = activeDay;
      const input = document.getElementById("inputRenameDay");
      input.value = activeDay;
      modal.classList.remove("hidden");
      modal.classList.add("flex");
      input.focus();
      input.select();
    }

    function closeRenameDayModal() {
      const modal = document.getElementById("renameDayModal");
      modal.classList.add("hidden");
      modal.classList.remove("flex");
    }

    function saveRenamedDay(e) {
      e.preventDefault();
      const newName = document.getElementById("inputRenameDay").value.trim();
      if (!newName || newName === activeDay) {
        closeRenameDayModal();
        return;
      }
      if (splitData[newName]) {
        showToast(`"${newName}" already exists.`, "error");
        return;
      }

      const updatedSplit = {};
      Object.keys(splitData).forEach(key => {
        if (key === activeDay) {
          updatedSplit[newName] = splitData[key];
        } else {
          updatedSplit[key] = splitData[key];
        }
      });

      splitData = updatedSplit;
      activeDay = newName;
      saveSplitData();
      closeRenameDayModal();
      renderTabs();
      renderExercises();
      showToast(`Renamed day to "${newName}"`);
    }

    function resetToDefaultSplit() {
      splitData = JSON.parse(JSON.stringify(DEFAULT_SPLIT_DATA));
      activeDay = Object.keys(splitData)[0];
      saveSplitData();
      renderTabs();
      renderExercises();
      showToast("Restored default MUSTAFA PROGRAM split");
    }

    // EXPORT STANDALONE HTML FOR GITHUB / VERCEL DEPLOYMENT
    // =====================================================================
    function exportStandaloneHTML() {
      const currentDocHTML = "<!DOCTYPE html>\n" + document.documentElement.outerHTML;
      const serializedData = JSON.stringify(splitData, null, 6);

      const updatedHTML = currentDocHTML.replace(
        /const DEFAULT_SPLIT_DATA = \{[\s\S]*?\n    \};/,
        `const DEFAULT_SPLIT_DATA = ${serializedData};`
      );

      const textArea = document.createElement("textarea");
      textArea.value = updatedHTML;
      textArea.style.position = "fixed";
      textArea.style.left = "-9999px";
      document.body.appendChild(textArea);
      textArea.select();

      try {
        document.execCommand("copy");
        showToast("MUSTAFA PROGRAM HTML copied! Ready to paste into GitHub/Vercel.");
      } catch (err) {
        showToast("Could not copy automatically. Check browser permissions.", "error");
      } finally {
        document.body.removeChild(textArea);
      }
    }
